### **Example: Lease Sentinel Flow**

```python
# agents/lease_sentinel.py
from langgraph.graph import StateGraph
from vertexai.generative_models import GenerativeModel

class LeaseSentinelAgent:
    def __init__(self):
        self.model = GenerativeModel("gemini-2.0-flash")
        self.graph = self._build_workflow()
    
    def _build_workflow(self):
        workflow = StateGraph()
        
        # Multi-step analysis pipeline
        workflow.add_node("extract_clauses", self.extract_clauses)
        workflow.add_node("risk_analysis", self.analyze_risks)
        workflow.add_node("compare_market", self.compare_to_market)
        workflow.add_node("generate_report", self.create_report)
        
        # Define flow
        workflow.set_entry_point("extract_clauses")
        workflow.add_edge("extract_clauses", "risk_analysis")
        workflow.add_edge("risk_analysis", "compare_market")
        workflow.add_edge("compare_market", "generate_report")
        
        return workflow.compile()
    
    async def analyze_lease(self, pdf_bytes: bytes):
        # OCR extraction
        text = await self.ocr_parser.parse(pdf_bytes)
        
        # Retrieve similar leases (hybrid search)
        similar_leases = await self.hybrid_retriever.search(
            query=text[:500],  # First 500 chars
            filters={"property_type": "residential"},
            top_k=5
        )
        
        # Run LangGraph workflow
        result = await self.graph.ainvoke({
            "lease_text": text,
            "market_data": similar_leases,
            "risk_threshold": 0.7
        })
        
        return result
```

    cached = redis_client.get(f"search:{hash(query)}")
    if cached:
        return cached
